1. PRIVACY-FIRST
   ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   ✓ 100% on-device computer vision (MediaPipe runs locally)
   ✓ NO frames sent to cloud (only metrics: 12 blinks/min, 75% posture)
   ✓ Edge AI architecture (all inference on user's machine)
   ✓ Compliance: GDPR-ready from day one

2. OFFLINE-FIRST
   ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   ✓ SQLite local database (works 100% offline)
   ✓ Sync queue (batches uploaded every 5 min when online)
   ✓ Idempotent uploads (UUID batch_id prevents duplicates)
   ✓ 7-day offline tolerance (2,016 batches = 100 KB)

3. SIMPLE STACK
   ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   ✓ PyQt6 (desktop app - one codebase for Windows/macOS)
   ✓ Supabase (auth + database + realtime - no custom backend)
   ✓ Next.js (dashboard - static site, free hosting)
   ✗ NO Kubernetes, Kafka, Redis, Microservices, etc.

4. CLIENT-HEAVY
   ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   ✓ Alert engine runs in desktop app (zero server load)
   ✓ Personal baseline calculated locally
   ✓ Calendar integration via OS APIs (no cloud dependency)
   ✓ Flow state detection client-side

5. TIME-SERIES OPTIMIZED
   ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   ✓ TimescaleDB extension (auto-partitioning by time)
   ✓ Continuous aggregates (pre-computed rollups)
   ✓ Retention policies (auto-delete > 30 days)
   ✓ Query rollups, not raw data (30 rows vs 25M rows)
